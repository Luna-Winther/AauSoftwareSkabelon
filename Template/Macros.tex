\newcommand{\setimagelabel}[1]{
\StrBehind{#1}{/}[\imagename]
\StrBefore{\imagename}{.}[\imagename]
\label{fig:\imagename}
}

\newcommand{\newimage}[4]{
    \IfStrEq{#1}{center}{
        \begin{figure}[htbp]
            \centering
            \includegraphics[width=#2\textwidth]{#3}
            \caption{#4}
            \setimagelabel{#3}
        \end{figure}%
    }{
        \IfStrEq{#1}{left}{
            \begin{wrapfigure}{l}{#2\textwidth}
                \includegraphics[width=\linewidth]{#3}
                \caption{#4}
                \setimagelabel{#3}
            \end{wrapfigure}
        }{
            \IfStrEq{#1}{right}{
                \begin{wrapfigure}{r}{#2\textwidth}
                    \includegraphics[width=\linewidth]{#3}
                    \caption{#4}
                    \setimagelabel{#3}
                \end{wrapfigure}
            }{}
        }
    }
}

\NewDocumentCommand{\AAUtable}{m +m m}
{
    \begin{table}
        \centering

        \str_case:nnF { #1 }
        {
            {noheader}
            {
                \begin{tblr}{
                    row{odd}  = {bg=tablebg},
                    row{even} = {bg=tablealt},
                    hline{1-Z} = {0.4pt},
                    vline{1-Z} = {0.4pt},
                }
                    #2
                \end{tblr}
            }

            {header}
            {
                \begin{tblr}{
                    row{1} = {
                        bg=tableheader,
                        font=\bfseries,
                    },
                    row{2} = {bg=tablebg},
                    row{3,5,...} = {bg=tablealt},
                    hline{1} = {1.2pt},
                    hline{2} = {0.8pt},
                    hline{3-Z} = {0.4pt},
                    vline{1-Z} = {0.4pt},
                }
                    #2
                \end{tblr}
            }

            {side}
            {
                \begin{tblr}{
                    row{1} = {
                        bg=tableheader,
                        font=\bfseries,
                    },
                    column{1} = {
                        bg=tableheader,
                        font=\bfseries,
                    },
                    row{2} = {bg=tablebg},
                    row{3,5,...} = {bg=tablealt},
                    hline{1} = {1.2pt},
                    hline{2} = {0.8pt},
                    hline{3-Z} = {0.4pt},
                    vline{1} = {1.2pt},
                    vline{2-Z} = {0.4pt},
                }
                    #2
                \end{tblr}
            }
        }
        {
            \PackageError{AAUtable}
                {Unknown~table~mode~'#1'}
                {Use~'noheader',~'header',~or~'side'.}
        }

        \tl_if_blank:nF { #3 }
        {
            \caption{#3}
        }

    \end{table}
}
